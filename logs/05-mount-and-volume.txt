########## 1. 바인드 마운트 - 호스트 변경이 재빌드 없이 반영되는가 ##########

# -v <호스트 절대경로>:<컨테이너 경로>:ro  (ro = 컨테이너는 읽기만 가능)

# 호스트 경로는 반드시 절대경로여야 한다. 상대경로를 쓰면 Docker 가 볼륨 이름으로 오해한다.

$ docker run -d -p 8083:80 -v /home/wakeuponce/codyssey/app:/usr/share/nginx/html:ro --name codyssey-web-bind codyssey-web:1.0
224f71a129ab00dffcfcd20fb76ed344c698c222762be558fdb71b4243ec59a2

$ sleep 2

$ docker ps --filter name=codyssey-web-bind --format 'table {{.Names}}\t{{.Ports}}\t{{.Status}}'
NAMES               PORTS                                     STATUS
codyssey-web-bind   0.0.0.0:8083->80/tcp, [::]:8083->80/tcp   Up 2 seconds (health: starting)

# 마운트가 실제로 걸렸는지 확인

$ docker inspect codyssey-web-bind --format '{{json .Mounts}}'
[{"Type":"bind","Source":"/home/wakeuponce/codyssey/app","Destination":"/usr/share/nginx/html","Mode":"ro","RW":false,"Propagation":"rprivate"}]

# [변경 전] 호스트 파일 내용

$ grep 'class="marker"' app/index.html
    <p class="marker" id="marker">SOURCE: image (빌드 시점에 이미지로 구운 파일)</p>

# [변경 전] 컨테이너가 서빙하는 내용 - 위와 같아야 한다.

$ curl -sS http://localhost:8083/ | grep 'class="marker"'
    <p class="marker" id="marker">SOURCE: image (빌드 시점에 이미지로 구운 파일)</p>

# 이제 호스트 파일만 수정한다. 재빌드(docker build)도, 재시작(docker restart)도 하지 않는다.

$ sed -i 's|SOURCE: image (빌드 시점에 이미지로 구운 파일)|SOURCE: host bind-mount (호스트에서 수정됨, 재빌드 없음)|' app/index.html

# [변경 후] 호스트 파일 내용

$ grep 'class="marker"' app/index.html
    <p class="marker" id="marker">SOURCE: host bind-mount (호스트에서 수정됨, 재빌드 없음)</p>

# [변경 후] 컨테이너 응답 - 재빌드 없이 즉시 바뀌었다.

$ curl -sS http://localhost:8083/ | grep 'class="marker"'
    <p class="marker" id="marker">SOURCE: host bind-mount (호스트에서 수정됨, 재빌드 없음)</p>

# 결정적 비교: 8080 은 바인드 마운트 없이 '이미지에 구운' 파일을 서빙 중이라 그대로다.

# 즉 이미지는 빌드 시점에 고정된 스냅샷이고, 바인드 마운트는 런타임에 호스트를 비추는 창이다.

$ curl -sS http://localhost:8080/ | grep 'class="marker"'
    <p class="marker" id="marker">SOURCE: image (빌드 시점에 이미지로 구운 파일)</p>

$ curl -sS http://localhost:8083/ | grep 'class="marker"'
    <p class="marker" id="marker">SOURCE: host bind-mount (호스트에서 수정됨, 재빌드 없음)</p>

# ro(읽기 전용)로 걸었으므로 컨테이너가 이 파일을 고치는 것은 거부된다.

$ docker exec codyssey-web-bind sh -c 'echo hacked > /usr/share/nginx/html/index.html'
sh: can't create /usr/share/nginx/html/index.html: Read-only file system
[exit=1]

# [복구] 원본 파일로 되돌린다.

$ cp /tmp/index.html.orig app/index.html

$ grep 'class="marker"' app/index.html
    <p class="marker" id="marker">SOURCE: image (빌드 시점에 이미지로 구운 파일)</p>

$ curl -sS http://localhost:8083/ | grep 'class="marker"'
    <p class="marker" id="marker">SOURCE: image (빌드 시점에 이미지로 구운 파일)</p>

$ docker rm -f codyssey-web-bind

########## 2. 볼륨 - 컨테이너를 지워도 데이터가 남는가 ##########

# 볼륨 생성. 바인드 마운트와 달리 호스트 경로를 우리가 정하지 않는다 - Docker 가 관리한다.

$ docker volume create codyssey-data
codyssey-data

$ docker volume ls
DRIVER    VOLUME NAME
local     codyssey-data

$ docker volume inspect codyssey-data
[
    {
        "CreatedAt": "2026-07-29T15:37:51+09:00",
        "Driver": "local",
        "Labels": null,
        "Mountpoint": "/var/lib/docker/volumes/codyssey-data/_data",
        "Name": "codyssey-data",
        "Options": null,
        "Scope": "local"
    }
]

# [1단계] 볼륨을 붙인 컨테이너에서 데이터를 쓴다.

$ docker run -d --name vol-test -v codyssey-data:/data ubuntu:24.04 sleep 300
32052dd90bf38e718257fd4bc0295f68b6a9743b93697d3fb6c4621d78ffc0ad

$ docker exec vol-test bash -c "echo 'codyssey 볼륨 영속성 테스트' > /data/hello.txt"

# $(hostname) 은 반드시 '컨테이너 안에서' 확장돼야 한다. 작은따옴표로 감싸 호스트 셸이

# 먼저 치환해버리는 것을 막는다. (안 그러면 호스트 이름이 기록된다 - 트러블슈팅 #3)

$ docker exec vol-test sh -c 'echo "작성 컨테이너 ID: $(hostname)" >> /data/hello.txt'

$ docker exec vol-test cat /data/hello.txt
codyssey 볼륨 영속성 테스트
작성 컨테이너 ID: 32052dd90bf3

# [2단계] 컨테이너 삭제 전 상태

$ docker ps -a --filter name=vol-test --format 'table {{.Names}}\t{{.Status}}'
NAMES      STATUS
vol-test   Up 1 second

# [3단계] 컨테이너를 강제 삭제한다. -f 는 실행 중이어도 지운다.

$ docker rm -f vol-test
vol-test

$ docker ps -a --filter name=vol-test --format 'table {{.Names}}\t{{.Status}}'
NAMES     STATUS

# [4단계] 컨테이너는 사라졌지만 볼륨은 그대로 있다.

$ docker volume ls
DRIVER    VOLUME NAME
local     codyssey-data

# [5단계] 새 컨테이너에 같은 볼륨을 붙여 데이터가 살아있는지 확인한다.

$ docker run -d --name vol-test2 -v codyssey-data:/data ubuntu:24.04 sleep 300
a4c36d878c62aeb8f96200f0e9cda9cf78ccce879696d77bdbf92393acf98934

$ docker exec vol-test2 cat /data/hello.txt
codyssey 볼륨 영속성 테스트
작성 컨테이너 ID: 32052dd90bf3

# 위 출력의 '작성 컨테이너 ID' 는 지금 컨테이너가 아니라 이미 삭제된 컨테이너의 것이다.

$ docker exec vol-test2 hostname
a4c36d878c62


########## 3. 대조군 - 볼륨 없이 컨테이너 내부에만 쓰면 어떻게 되는가 ##########

# 볼륨을 붙이지 않고 컨테이너 쓰기 가능 레이어에만 파일을 쓴다.

$ docker run -d --name novol-test ubuntu:24.04 sleep 300
fb475cb798c9c60cebe073b68e6aab33333e2b8b43d9e3e72bdc3669187dbdcb

$ docker exec novol-test bash -c "echo '이 데이터는 사라진다' > /data-nowhere.txt && cat /data-nowhere.txt"
이 데이터는 사라진다

# 컨테이너를 지운다.

$ docker rm -f novol-test
novol-test

# 같은 이미지로 새 컨테이너를 띄우면 그 파일은 없다. 쓰기 레이어가 컨테이너와 함께 삭제됐기 때문이다.

$ docker run -d --name novol-test2 ubuntu:24.04 sleep 300
7333ef8c0fa4e7b52db93f23e07181eec4140855238db5feb2f6769cb875ff38

$ docker exec novol-test2 cat /data-nowhere.txt
cat: /data-nowhere.txt: No such file or directory
[exit=1]
