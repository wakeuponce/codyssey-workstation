########## 1. Git 사용자 정보 및 기본 브랜치 설정 ##########

# GitHub 계정에서 로그인명과 숫자 ID 를 읽어 noreply 주소를 구성한다.

# 형식: <숫자ID>+<로그인명>@users.noreply.github.com

$ git config --global user.name 'wakeuponce'

$ git config --global user.email '310057122+wakeuponce@users.noreply.github.com'

# init.defaultBranch - 새 저장소의 기본 브랜치 이름. master 대신 main 을 쓴다.

$ git config --global init.defaultBranch main

# 전역 설정 전체

$ git config --global --list

init.defaultbranch=main

########## 7. [보너스] GitHub SSH 인증 ##########

# gh auth login 과정에서 ed25519 키가 생성되어 GitHub 계정에 등록되었다.

# HTTPS 는 매 푸시마다 토큰이 필요한 반면, SSH 는 키 쌍으로 인증한다.

# 개인키는 소유자만 읽을 수 있어야 한다(600). 권한이 느슨하면 ssh 가 사용을 거부한다.

$ ls -l ~/.ssh/id_ed25519 ~/.ssh/id_ed25519.pub
-rw------- 1 wakeuponce wakeuponce 387 Jul 29 16:49 /home/wakeuponce/.ssh/id_ed25519
-rw-r--r-- 1 wakeuponce wakeuponce  82 Jul 29 16:49 /home/wakeuponce/.ssh/id_ed25519.pub

$ stat -c '%A (%a) %n' ~/.ssh/id_ed25519
-rw------- (600) /home/wakeuponce/.ssh/id_ed25519

# GitHub 에 등록된 공개키 목록 (공개키는 비밀이 아니므로 그대로 기록한다)

$ gh ssh-key list 2>/dev/null | cut -c1-60 || echo '(조회 권한 없음)'
GitHub CLI	ssh-ed25519 AAAAC3NzaC1lZDI1NTE5AAAAIEiFiVneWcmrf

# 실제 인증이 되는지 확인. GitHub 은 셸 접속을 주지 않으므로 이 메시지가 정상 응답이다.

$ ssh -T git@github.com 2>&1 | head -2; true
Hi wakeuponce! You've successfully authenticated, but GitHub does not provide shell access.

# remote 가 git@ 로 시작하면 SSH 프로토콜을 쓰는 것이다.

$ git remote get-url origin
git@github.com:wakeuponce/codyssey-workstation.git

# 개인키가 저장소에 섞여 들어가지 않는지 확인 (.gitignore 로 차단)

$ git check-ignore -v id_ed25519 || echo '(경로상 해당 없음 - 키는 ~/.ssh 에 있고 저장소 밖이다)'
