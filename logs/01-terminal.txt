

########## 0. 실습 준비 - 깨끗한 상태에서 시작 ##########

$ rm -rf /home/wakeuponce/codyssey-lab


########## 1. 현재 위치 확인 (pwd) ##########

# pwd 는 'print working directory' - 지금 셸이 서 있는 절대경로를 출력한다.

$ pwd
/home/wakeuponce/codyssey


########## 2. 디렉토리 생성 (mkdir -p) ##########

# -p 는 중간 경로가 없으면 같이 만들고, 이미 있어도 에러를 내지 않는다.

$ mkdir -p /home/wakeuponce/codyssey-lab/project/src

$ mkdir -p /home/wakeuponce/codyssey-lab/backup


########## 3. 이동 (cd) + 절대경로 vs 상대경로 ##########

# 절대경로: / 에서 시작하는 완전한 주소. 현재 위치와 무관하게 항상 같은 곳을 가리킨다.

$ cd /home/wakeuponce/codyssey-lab/project && pwd
/home/wakeuponce/codyssey-lab/project

# 상대경로: 현재 위치를 기준으로 한 주소. '.' = 현재, '..' = 상위 디렉토리.

$ cd /home/wakeuponce/codyssey-lab/project/src && pwd && cd .. && pwd
/home/wakeuponce/codyssey-lab/project/src
/home/wakeuponce/codyssey-lab/project

# 같은 대상을 절대경로와 상대경로 두 방식으로 가리켜 본다 (출력이 동일해야 한다).

$ cd /home/wakeuponce/codyssey-lab/project/src && ls -d /home/wakeuponce/codyssey-lab/backup
/home/wakeuponce/codyssey-lab/backup

$ cd /home/wakeuponce/codyssey-lab/project/src && ls -d ../../backup
../../backup


########## 4. 빈 파일 생성 (touch) ##########

# touch 는 파일이 없으면 0바이트 파일을 만들고, 있으면 수정시각만 갱신한다.

$ cd /home/wakeuponce/codyssey-lab/project && touch empty.txt && ls -l empty.txt
-rw-r--r-- 1 wakeuponce wakeuponce 0 Jul 29 15:27 empty.txt


########## 5. 파일 생성 및 내용 확인 (cat) ##########

$ cd /home/wakeuponce/codyssey-lab/project && printf 'hello codyssey\nline2\n' > note.txt

$ cd /home/wakeuponce/codyssey-lab/project && cat note.txt
hello codyssey
line2


########## 6. 목록 확인 - 숨김 파일 포함 (ls -la) ##########

# '.' 으로 시작하는 파일은 기본 ls 에 안 보인다. -a 를 붙여야 보인다.

$ cd /home/wakeuponce/codyssey-lab/project && touch .hidden-config

$ cd /home/wakeuponce/codyssey-lab/project && ls -l
total 8
-rw-r--r-- 1 wakeuponce wakeuponce    0 Jul 29 15:27 empty.txt
-rw-r--r-- 1 wakeuponce wakeuponce   21 Jul 29 15:27 note.txt
drwxr-xr-x 2 wakeuponce wakeuponce 4096 Jul 29 15:27 src

$ cd /home/wakeuponce/codyssey-lab/project && ls -la
total 16
drwxr-xr-x 3 wakeuponce wakeuponce 4096 Jul 29 15:27 .
drwxr-xr-x 4 wakeuponce wakeuponce 4096 Jul 29 15:27 ..
-rw-r--r-- 1 wakeuponce wakeuponce    0 Jul 29 15:27 .hidden-config
-rw-r--r-- 1 wakeuponce wakeuponce    0 Jul 29 15:27 empty.txt
-rw-r--r-- 1 wakeuponce wakeuponce   21 Jul 29 15:27 note.txt
drwxr-xr-x 2 wakeuponce wakeuponce 4096 Jul 29 15:27 src


########## 7. 복사 (cp) ##########

# 파일 복사. 디렉토리를 통째로 복사할 때는 -r 이 필요하다.

$ cd /home/wakeuponce/codyssey-lab/project && cp note.txt /home/wakeuponce/codyssey-lab/backup/note-copy.txt

$ ls -l /home/wakeuponce/codyssey-lab/backup
total 4
-rw-r--r-- 1 wakeuponce wakeuponce 21 Jul 29 15:27 note-copy.txt

$ cp -r /home/wakeuponce/codyssey-lab/project /home/wakeuponce/codyssey-lab/backup/project-snapshot

$ ls -R /home/wakeuponce/codyssey-lab/backup
/home/wakeuponce/codyssey-lab/backup:
note-copy.txt
project-snapshot

/home/wakeuponce/codyssey-lab/backup/project-snapshot:
empty.txt
note.txt
src

/home/wakeuponce/codyssey-lab/backup/project-snapshot/src:


########## 8. 이동 / 이름 변경 (mv) ##########

# mv 는 '옮기기'와 '이름 바꾸기'가 같은 명령이다. 목적지가 새 이름이면 rename 이 된다.

$ cd /home/wakeuponce/codyssey-lab/project && mv note.txt renamed.txt && ls -l
total 8
-rw-r--r-- 1 wakeuponce wakeuponce    0 Jul 29 15:27 empty.txt
-rw-r--r-- 1 wakeuponce wakeuponce   21 Jul 29 15:27 renamed.txt
drwxr-xr-x 2 wakeuponce wakeuponce 4096 Jul 29 15:27 src

$ cd /home/wakeuponce/codyssey-lab/project && mv renamed.txt src/renamed.txt && ls -l src
total 4
-rw-r--r-- 1 wakeuponce wakeuponce 21 Jul 29 15:27 renamed.txt


########## 9. 삭제 (rm) ##########

# rm 은 휴지통을 거치지 않는다. -r 은 디렉토리 재귀 삭제.

$ cd /home/wakeuponce/codyssey-lab/project && rm empty.txt && ls -l
total 4
drwxr-xr-x 2 wakeuponce wakeuponce 4096 Jul 29 15:27 src

$ rm -r /home/wakeuponce/codyssey-lab/backup/project-snapshot && ls /home/wakeuponce/codyssey-lab/backup
note-copy.txt
